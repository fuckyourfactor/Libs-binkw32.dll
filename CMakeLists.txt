cmake_minimum_required(VERSION 3.16)
project(bink_open VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(BINK_WITH_FFMPEG "Enable Bink video and audio decoders through FFmpeg" OFF)
option(BINK_FFMPEG_STATIC "Link FFmpeg libraries statically (single-file deployment)" OFF)
if(BINK_FFMPEG_STATIC AND NOT BINK_WITH_FFMPEG)
  message(FATAL_ERROR "BINK_FFMPEG_STATIC requires BINK_WITH_FFMPEG=ON")
endif()
option(BINK_BUILD_TESTS "Build unit and ABI/export verification tests" ON)
set(BK_SOURCES src/bink_api.c src/container.c src/pixel.c src/audio_mix.c src/sound_driver.c src/surface.c src/trace.c)
if(WIN32)
  list(APPEND BK_SOURCES src/platform_win32.c src/audio_win32.c)
else()
  list(APPEND BK_SOURCES src/platform_posix.c src/audio_none.c)
endif()
if(BINK_WITH_FFMPEG)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)
  list(APPEND BK_SOURCES src/decoder_ffmpeg.c)
else()
  list(APPEND BK_SOURCES src/decoder_none.c)
endif()
if(WIN32)

  set(BINK_MINGW_PUBLIC_NAME_PREFIX "single" CACHE STRING
      "GNU ld public export prefix (single, double, none)")
  set_property(CACHE BINK_MINGW_PUBLIC_NAME_PREFIX PROPERTY STRINGS single double none)
  if(NOT BINK_MINGW_PUBLIC_NAME_PREFIX MATCHES "^(single|double|none)$")
    message(FATAL_ERROR "Unknown BINK_MINGW_PUBLIC_NAME_PREFIX")
  endif()

  if(MINGW)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(BINK_EXPORT_DEF "${CMAKE_CURRENT_BINARY_DIR}/binkw32-mingw.def")
    execute_process(
      COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tools/make_mingw_def.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/binkw32.def"
        "${CMAKE_CURRENT_SOURCE_DIR}/tools/exports.json"
        "${BINK_EXPORT_DEF}"
        "--public-prefix" "${BINK_MINGW_PUBLIC_NAME_PREFIX}"
      RESULT_VARIABLE BINK_DEF_RESULT
      OUTPUT_VARIABLE BINK_DEF_OUTPUT
      ERROR_VARIABLE BINK_DEF_ERROR
    )
    if(NOT BINK_DEF_RESULT EQUAL 0)
      message(FATAL_ERROR "Cannot generate MinGW exports: ${BINK_DEF_ERROR}")
    endif()
    message(STATUS "${BINK_DEF_OUTPUT}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
      "${CMAKE_CURRENT_SOURCE_DIR}/src/binkw32.def"
      "${CMAKE_CURRENT_SOURCE_DIR}/tools/exports.json"
      "${CMAKE_CURRENT_SOURCE_DIR}/tools/make_mingw_def.py")
  else()
    set(BINK_EXPORT_DEF "${CMAKE_CURRENT_SOURCE_DIR}/src/binkw32.def")
  endif()
  add_library(binkw32 SHARED ${BK_SOURCES} "${BINK_EXPORT_DEF}")
  set_target_properties(binkw32 PROPERTIES PREFIX "" OUTPUT_NAME "binkw32")
  target_link_libraries(binkw32 PRIVATE gdi32 user32 kernel32 winmm)
  if(MINGW)

    target_link_options(binkw32 PRIVATE "-Wl,--disable-stdcall-fixup")
  endif()
else()
  add_library(binkw32 SHARED ${BK_SOURCES})
  set_target_properties(binkw32 PROPERTIES OUTPUT_NAME "bink_open_test")
endif()
target_include_directories(binkw32 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include PRIVATE src)
target_compile_definitions(binkw32 PRIVATE BINK_BUILD)
if(BINK_WITH_FFMPEG)
  target_compile_definitions(binkw32 PRIVATE BINK_WITH_FFMPEG=1)
  if(BINK_FFMPEG_STATIC)

    if(NOT AV_STATIC_LDFLAGS)
      message(FATAL_ERROR "pkg-config returned no --static linker flags for FFmpeg")
    endif()
    foreach(required avformat avcodec avutil swscale swresample)
      if(NOT "${AV_STATIC_LIBRARIES}" MATCHES "(^|;)${required}(;|$)")
        message(FATAL_ERROR "Static FFmpeg link is missing ${required}: ${AV_STATIC_LDFLAGS}")
      endif()
    endforeach()
    message(STATUS "Static FFmpeg link flags: ${AV_STATIC_LDFLAGS}")
    target_include_directories(binkw32 PRIVATE ${AV_INCLUDE_DIRS})
    if(AV_STATIC_CFLAGS_OTHER)
      target_compile_options(binkw32 PRIVATE ${AV_STATIC_CFLAGS_OTHER})
    endif()

    if(WIN32 AND MINGW)

      target_link_libraries(binkw32 PRIVATE
        "-Wl,--start-group" ${AV_STATIC_LDFLAGS} "-Wl,--end-group"
        gdi32 user32 kernel32 winmm)
      target_link_options(binkw32 PRIVATE -static -static-libgcc)
    else()
      target_link_libraries(binkw32 PRIVATE ${AV_STATIC_LDFLAGS})
    endif()
  else()
    target_link_libraries(binkw32 PRIVATE PkgConfig::AV)
  endif()
endif()

 find_package(Python3 COMPONENTS Interpreter)
 if(Python3_Interpreter_FOUND)
   add_test(NAME export_manifest COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_def.py ${CMAKE_CURRENT_SOURCE_DIR}/src/binkw32.def ${CMAKE_CURRENT_SOURCE_DIR}/tools/exports.json)
   add_test(NAME mingw_export_aliases COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_mingw_def.py)
   add_test(NAME pe_export_directory COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pe_exports.py)
   add_test(NAME win32_runtime_imports COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_win32_imports.py)
