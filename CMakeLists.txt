cmake_minimum_required(VERSION 3.16)
project(bink_open VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(BINK_WITH_FFMPEG "Enable Bink video and audio decoders through FFmpeg" OFF)
option(BINK_FFMPEG_STATIC "Link FFmpeg libraries statically (single-file deployment)" OFF)
if(BINK_FFMPEG_STATIC AND NOT BINK_WITH_FFMPEG)
  message(FATAL_ERROR "BINK_FFMPEG_STATIC requires BINK_WITH_FFMPEG=ON")
endif()
option(BINK_BUILD_TESTS "Build unit and ABI/export verification tests" ON)
set(BK_SOURCES src/bink_api.c src/container.c src/pixel.c src/audio_mix.c src/sound_driver.c src/surface.c)
if(WIN32)
  list(APPEND BK_SOURCES src/platform_win32.c src/audio_win32.c)
else()
  list(APPEND BK_SOURCES src/platform_posix.c src/audio_none.c)
endif()
if(BINK_WITH_FFMPEG)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)
  list(APPEND BK_SOURCES src/decoder_ffmpeg.c)
else()
  list(APPEND BK_SOURCES src/decoder_none.c)
endif()
if(WIN32)
  add_library(binkw32 SHARED ${BK_SOURCES} src/binkw32.def)
  set_target_properties(binkw32 PROPERTIES PREFIX "" OUTPUT_NAME "binkw32")
  target_link_libraries(binkw32 PRIVATE gdi32 user32 kernel32 winmm)
  if(MINGW)
    target_link_options(binkw32 PRIVATE "-Wl,--enable-stdcall-fixup")
  endif()
else()
  add_library(binkw32 SHARED ${BK_SOURCES})
  set_target_properties(binkw32 PROPERTIES OUTPUT_NAME "bink_open_test")
endif()
target_include_directories(binkw32 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include PRIVATE src)
target_compile_definitions(binkw32 PRIVATE BINK_BUILD)
if(BINK_WITH_FFMPEG)
  if(BINK_FFMPEG_STATIC)
    # PkgConfig::AV uses pkg-config --libs, which omits Libs.private.
    # A static FFmpeg link MUST also use pkg-config --static --libs:
    # those private libraries include platform dependencies.
    target_include_directories(binkw32 PRIVATE ${AV_INCLUDE_DIRS})
    target_link_directories(binkw32 PRIVATE ${AV_STATIC_LIBRARY_DIRS})
    target_link_libraries(binkw32 PRIVATE ${AV_STATIC_LINK_LIBRARIES})
    if(AV_STATIC_CFLAGS_OTHER)
      target_compile_options(binkw32 PRIVATE ${AV_STATIC_CFLAGS_OTHER})
    endif()
    if(AV_STATIC_LDFLAGS_OTHER)
      target_link_options(binkw32 PRIVATE ${AV_STATIC_LDFLAGS_OTHER})
    endif()
    if(WIN32 AND MINGW)
      target_link_options(binkw32 PRIVATE -static -static-libgcc)
    endif()
  else()
    target_link_libraries(binkw32 PRIVATE PkgConfig::AV)
  endif()
endif()
if(BINK_BUILD_TESTS)
 enable_testing()
 add_executable(bink_tests tests/test_core.c src/container.c)
 target_include_directories(bink_tests PRIVATE include src)
 target_link_libraries(bink_tests PRIVATE binkw32)
 add_test(NAME core COMMAND bink_tests)
 find_package(Python3 COMPONENTS Interpreter)
 if(Python3_Interpreter_FOUND)
   add_test(NAME export_manifest COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_def.py ${CMAKE_CURRENT_SOURCE_DIR}/src/binkw32.def ${CMAKE_CURRENT_SOURCE_DIR}/docs/exports.json)
 endif()
endif()

if(BINK_BUILD_TESTS)
 add_executable(pixel_tests tests/test_pixels.c src/pixel.c)
 target_include_directories(pixel_tests PRIVATE src)
 add_test(NAME pixels COMMAND pixel_tests)
endif()

if(BINK_BUILD_TESTS)
 add_executable(audio_mix_tests tests/test_audio_mix.c src/audio_mix.c)
 target_include_directories(audio_mix_tests PRIVATE src)
 add_test(NAME audio_mix COMMAND audio_mix_tests)
endif()

if(BINK_BUILD_TESTS)
 add_executable(surface_tests tests/test_surface.c src/surface.c)
 target_include_directories(surface_tests PRIVATE include src)
 add_test(NAME surface_mapping COMMAND surface_tests)
endif()

if(BINK_BUILD_TESTS)
 add_executable(sound_api_tests tests/test_sound_api.c)
 target_include_directories(sound_api_tests PRIVATE include)
 target_link_libraries(sound_api_tests PRIVATE binkw32)
 add_test(NAME sound_driver_api COMMAND sound_api_tests)
endif()

# A real-media integration harness. It is not an automated green test:
# successful decoding requires a legally obtained Bink1 movie and FFmpeg.
add_executable(bink_probe tools/probe_bink.c)
target_include_directories(bink_probe PRIVATE include)
target_link_libraries(bink_probe PRIVATE binkw32)
